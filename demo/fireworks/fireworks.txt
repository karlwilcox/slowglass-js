# Fireworks demo
# All images created by Affinty App generative AI
# call from URL /scripts/fireworks.txt

display width 800
display height 450

setup
    gravity 50
    # Create our stage...
    from /demo/fireworks
    load starry-sky.png as starry-sky
    use starry-sky as background
    load crowd-of-people.png as crowd
    use crowd as foreground
    load tree1.png as left-tree
    load tree2.png as right-tree
    sprite create left-tree from left-tree
    sprite create right-tree from right-tree
    darken left-tree to 70
    darken right-tree to 70
    place left-tree at 20 200 depth 250
    place right-tree at 780 200 depth  251

every 3 seconds
    wait for (1 + (4 * $CHANCE)) seconds
    start firework1

scene firework1
   begin
        from /demo/fireworks
        load firework-1.png as firework1 cells 4 by 1
        make launch-site be (600 * $CHANCE)
        # launch from anywhere between 100 and 600 but tend to point towards the centre
        make start-angle be ((((${launch-site} / 600) * 50) - 25) * -1)
        make launch-angle be (${start-angle} + (-20 + (40 * $CHANCE)))
        make launch-speed be (120 + (80 * $CHANCE))
        make big-size be (100 + (2 * $PERCENT))
        choose tint from red lightcyan lawngreen yellow orangered gold tomato
        sprite create firework1 from firework1
        place firework1 at (100 + ${launch-site}) 450 depth 220 size 50 50
        tint firework1 $tint
        rotate firework1 $launch-angle
        launch firework1 at ${launch-angle} with force ${launch-speed}
        resize firework1 to 100 100 in 2 seconds
        then
        advance firework1 by 1
        darken main:left-tree to 50 in 1 second
        darken main:right-tree to 50 in 1 second
        resize firework1 to ${big-size} ${big-size} in 1 seconds
        then 
        advance firework1 by 1
        // scale firework1 by 120% in 1 second
        then
        advance firework1 by 1
        darken main:left-tree to 70 in 1 second
        darken main:right-tree to 70 in 1 second
        // fade firework1 to 0 in 2 second
    
    after 20 seconds
        delete scene

    when ${firework1.landed}
        log landed!
        delete scene
end scene
